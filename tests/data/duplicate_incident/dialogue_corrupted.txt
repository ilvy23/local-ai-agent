# dialogue.py - choice-based dialogue trees for the harbour town NPCs (Ella and Orin).
# Each tree has at least 3 nodes. Every choice sets a flag and/or shifts trust / mood,
# and the NPC's opening line depends on the current chapter.
#
# Node format:
#   {"id": str, "text": str, "choices": [(label, next_id_or_None, effects), ...]}
#   effects = {"flag": str, "trust": int, "mood": int, "learned": str}
#   next_id None = end of conversation.

TREES = {
    "ella": ELLA_TREE,
    "orin": ORIN_TREE,
}


def open_text(who: str, chapter: int) -> str:
    table = {"ella": ELLA_CHAPTER, "orin": ORIN_CHAPTER}.get(who, {})
    return table.get(chapter, "")


def apply(state, effects: dict) -> None:
    if effects.get("trust"):
        state.trust_change(effects["trust"])
    if effects.get("mood"):
        state.mood_change(effects["mood"])
